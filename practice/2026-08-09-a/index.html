拓展为 HTTP 服务器

简易 TCP 服务器:

#include <stdio.h>
#include <unistd.h>
#include <string.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <netdb.h>

#define PORT "3490"


int main(void) {
    int listenfd, connfd;
    struct addrinfo hints, *servinfo, *p;
    struct sockaddr_storage their_addr;
    socklen_t sin_size;

    memset(&hints, 0, sizeof hints);
    hints.ai_family = AF_UNSPEC;
    hints.ai_socktype = SOCK_STREAM;
    hints.ai_flags = AI_PASSIVE;
    
    int rv = getaddrinfo(NULL, PORT, &hints, &servinfo);
    if (rv != 0) {
        fprintf(stderr, "getaddrinfo: %s\n", gai_strerror(rv));
        return 1;
    }

    for (p = servinfo; p != NULL; p = p->ai_next) {
        if ((listenfd = socket(p->ai_family, p->ai_socktype, p->ai_protocol)) == -1) {
            perror("server: socket");
            continue;
        }

        if (bind(listenfd, p->ai_addr, p->ai_addrlen) == -1) {
            close(listenfd);
            perror("server: bind");
            continue;
        }

        break;
    }

    if (p == NULL) {
        fprintf(stderr, "server: failed to bind\n");
        return 2;
    }

    freeaddrinfo(servinfo);

    if (listen(listenfd, 5) == -1) {
        perror("listen");
        return 3;
    }

    while(1) {
        sin_size = sizeof their_addr;
        if ((connfd = accept(listenfd, (struct sockaddr *)&their_addr, &sin_size)) == -1) {
            perror("accept");
            continue;
        }

        if (send(connfd, "Hello world!", 12, 0) == -1) {
            perror("send");
        }

        close(connfd);
    }
}

作为传输层的 TCP 不管上层是什么协议, 只负责传 byte stream

TCP 提供可靠的 byte stream 通道, HTTP 则规定了: bytes 应该按照什么格式组织, 如何解析, 以及服务器应该如何回应

HTTP 版本

HTTP 是一组应用层协议, 不是一种固定的 wire format. 不同版本共享很多语义, 但在网络上传输时的编码方式并不一样:

版本请求在网络上的基本形式
HTTP/1.0、HTTP/1.1可读的文本请求行和请求头, 使用 CRLF 分隔
HTTP/2binary frames, stream 和 frame header
HTTP/3基于 QUIC 的 binary frames
  1. 客户端要表达什么请求, 例如使用什么 method, 请求哪个 target.
  2. 请求头提供了哪些元数据, 例如目标主机和 body 的长度.
  3. 服务器如何根据这些字段生成 response.

示例 HTTP/1.x 请求

HTTP/1.x (HTTP/1.0 和 HTTP/1.1) 的文本请求通常由以下部分组成:

请求行 + 请求头 + 空行 + 请求体

常见的 GET 请求:

GET /index.html HTTP/1.1\r\n
Host: example.com\r\n
Accept: text/html\r\n
\r\n

请求行:   GET /index.html HTTP/1.1\r\n
请求头:   Host: example.com\r\n
          Accept: text/html\r\n
空行:     \r\n
请求体:   这里没有 body
  • 请求行表达"用什么 method, 访问哪个 target, 使用哪个 HTTP version".
  • 每个请求头都是 name: value.
  • 空行表示请求头结束.
  • 请求体是可选的.

HTTP/1.1 只是上面请求行中的 version 字段. 换成 HTTP/1.0 后, 仍然可以用同样的基本结构表示请求; 但两个版本在 Host、持久连接和 body framing 等规则上并不完全相同.

获取完整的 HTTP/1.x 请求

TCP 只提供连续的 byte stream, 一次 recv() 可能只得到部分数据, 需要 recv() 后继续尝试解析直到获得完整请求数据

请求行和请求头以 \r\n 分隔, 空行 \r\n\r\n 表示请求头结束. 请求体是否存在以及它的长度, 由 HTTP/1.x 的 message framing 规则决定. 为了先理解 TCP 与 HTTP 的配合, 本例只处理 Content-Length 指定长度的请求体, 因此获取完整请求的过程是:

  1. 不断调用 recv(), 直到收到 \r\n\r\n
  2. 解析请求头中的 Content-Length
  3. 继续 recv(), 直到已经收到 Content-Length 个 body bytes

为避免每次扩容都复制整个 buffer, 直接使用固定大小的 buffer

函数设计:

sockfd + buffer + buffer capacity
              ↓
     recv_http1_request()
              ↓
完整请求 bytes + bytes length

buf 由调用方创建, capacity 表示 buf 的大小. 函数将完整请求写入 buf, 并返回请求的 byte 数量.

#include <errno.h>
#include <stdio.h>
#include <string.h>
#include <sys/socket.h>

#define REQUEST_BUFFER_SIZE 8192

/** 解析请求头中的 `Content-Length`, 到此需保证请求行和请求头已经获取完毕
 * - `buf`: 指向HTTP请求数据起始位置的指针
 * - `headers_end`: 指向HTTP头部结束位置的指针, 空行开始的位置
 *
 * Tips:
 * 1. static 表示此函数就只能在这个 .c 文件内部被调用
 * 2. char 是 C 中内存寻址的最小单位, 大小固定为 1 byte
 * 3. `char buf[] = "GET";` 表示"字符串", 内存中实际存储为 [71][69][84] ([G][E][T])
 * 4. strstr: 用于在字符串中查找子串的第一次出现
 * 5. sscanf: 用于从字符串中按照指定格式提取数据
 */
static size_t get_content_length(char *buf, char *headers_end) {
    char *line = strstr(buf, "\r\n");   // 在 buf 中查找第一个换行符 \r\n, 作为第一行的结束标记

    while (line != NULL && line < headers_end) {    // 循环遍历每一行, 直到到达头部结束位置或没有更多行
        line += 2;                                  // 跳过当前行的换行符, 移动到下一行的起始位置

        size_t length;
        if (sscanf(line, "Content-Length: %zu", &length) == 1) {    // 尝试解析当前行, 检查是否以"Content-Length:"开头
            return length;                                          // 如果成功匹配并提取到长度值, 立即返回该值
        }

        line = strstr(line, "\r\n");    // 查找下一行的换行符, 继续循环
    }

    return 0;
}

/** 从 sockfd 获取一个完整的 HTTP/1.x 请求
 * - `sockfd`: 已连接的套接字文件描述符
 * - `buf`: 用于存储 HTTP 请求数据的缓冲区, 指向数组起点
 * - `capacity`: 缓冲区总容量
 */
ssize_t recv_http1_request(int sockfd, char *buf, size_t capacity) {
    size_t received = 0;    // 记录当前已经接收到的数据长度

    if (capacity < 2) {     // 至少需要保留一个字节用于字符串结束符 '\0', 因此缓冲区容量必须至少为 2
        errno = EMSGSIZE;
        return -1;
    }

    while (received < capacity - 1) {   // 循环接收数据, 并始终为 '\0' 预留一个字节
        ssize_t bytes_received = recv(
            sockfd,
            buf + received,             // 将新数据追加到已有数据之后, 即指针指向已经接收到的数据后
            capacity - 1 - received,
            0
        );

        if (bytes_received == 0) {
            return 0;
        }

        if (bytes_received == -1) {
            if (errno == EINTR) {
                continue;
            }

            return -1;
        }

        received += bytes_received;
        buf[received] = '\0';

        char *headers_end = strstr(buf, "\r\n\r\n");    // 查找 HTTP/1.x 请求头和请求体之间的分隔符
        if (headers_end == NULL) {
            continue;
        }

        size_t headers_length = (headers_end - buf) + 4;    // [buf, ..., headers_end, "\r\n\r\n"]
        size_t body_length = get_content_length(buf, headers_end);

        if (body_length > capacity - 1 - headers_length) {  // 检查"请求头 + 请求体"是否能够放入当前缓冲区
            errno = EMSGSIZE;
            return -1;
        }

        size_t request_length = headers_length + body_length;   // 如果已经接收到完整的请求头和请求体, 则返回完整请求长度
        if (received >= request_length) {                       // received 可能大于 request_length
            return request_length;
        }
    }

    errno = EMSGSIZE;   // 缓冲区已满, 但仍未成功接收到完整请求
    return -1;
}

调用时, buf 中可能一次就包含请求头和部分请求体, 所以不能在第一次找到 \r\n\r\n 后立即返回. 只有 received >= request_length 时, 才说明请求行、请求头和请求体都已经完整收到:

char request[REQUEST_BUFFER_SIZE];
ssize_t request_length = recv_http1_request(
    connfd,
    request,
    sizeof request
);

if (request_length == -1) {
    perror("recv_http1_request");
} else if (request_length == 0) {
    /* 对端提前关闭连接 */
} else {
    /* request[0..request_length-1] 是完整的 HTTP/1.x 请求 */
}

这里的关键是 HTTP 自己提供消息边界, TCP 不提供消息边界. 对于没有 Content-Length 的请求, 本例将请求体长度视为 0, 收到请求头结束就可以交给后续 HTTP/1.x parser.

为了突出数据获取流程, 本例假设请求格式正确、header name 使用常见写法, 并且一个 TCP 连接只处理一个请求. 这只是 HTTP/1.x 的教学子集, 完整 HTTP/1.x 服务器还需要处理大小写不敏感的 header name、非法 Content-Length、Transfer-Encoding: chunked 和同一连接上的后续请求.

HTTP/1.x 请求解析函数

函数设计:

原始 bytes
    ↓
┌─────────────────────────────┐
│ HTTP/1.x Request Parser     │
│                             │
│ parse_request_line()        │ 请求行
│ parse_request_headers()     │ 请求头
│ parse_request_body()        │ 请求体
└─────────────────────────────┘
    ↓
struct http_request

函数接口可以设计为:

#include <stddef.h>

#define HTTP_MAX_HEADERS 32

struct http_header {
    char *name;
    char *value;
};

struct http_request {
    char *method;       // 请求方法, 例如 "GET"
    char *target;       // 请求目标, 例如 "/index.html"
    char *version;      // HTTP/1.x 版本, 例如 "HTTP/1.1"

    struct http_header headers[HTTP_MAX_HEADERS];
    size_t header_count;

    char *body;         // 请求体起始位置
    size_t body_length; // 请求体 byte 数量
};

/**
 * `buf` 中存放一条完整的 HTTP/1.x 请求, `length` 是实际请求长度
 * 成功返回 0, 请求格式错误返回 -1
 */
int parse_http1_request(char *buf, size_t length, struct http_request *request);

struct http_request 不复制字符串, 而是保存指向原始 buffer 的指针:

这样不需要 malloc() 和 free(). 但 parser 会把请求行和请求头中的分隔符改成 \0, 因此 buf 必须是可写的. request 中的所有指针都指向 buf, 只要 buf 被覆盖, 这些指针就不能再使用. body 不是 C 字符串, 访问它必须使用 body_length.

查找 HTTP/1.x 边界

虽然 HTTP/1.x 的请求行和请求头是文本, 这里仍然按照 [start, end) 的 byte 范围查找换行符, 不依赖 strstr() 或 strlen(). 这样可以明确哪些 bytes 属于当前请求. 这段代码只适用于 HTTP/1.x; HTTP/2 和 HTTP/3 应该先由对应的 framing layer 拆出字段, 不会经过这里:

#include <string.h>

/** 在 [start, end) 范围内查找第一个 `\r\n`
 * 成功时返回 `\r\n` 的起始位置, 即 `\r`
 */
static char *find_crlf(char *start, char *end) {
    while (end - start >= 2) {  // 至少剩余 2 个字节时, 才可能包含 "\r\n"
        if (start[0] == '\r' && start[1] == '\n') { // 当前位置的两个字符是否为 CRLF
            return start;
        }

        start++;
    }

    return NULL;
}

/** 在 [start, end) 中查找 HTTP/1.x 请求头结束标记 `\r\n\r\n`
 * 成功时返回 `\r\n\r\n` 的起始位置
 */
static char *find_headers_end(char *start, char *end) {
    while (end - start >= 4) {
        if (memcmp(start, "\r\n\r\n", 4) == 0) {    // 如果与 "\r\n\r\n" 完全一致, 则说明 HTTP Header 结束
            return start;
        }

        start++;
    }

    return NULL;
}

find_headers_end() 返回四个字节分隔符的第一个 \r:

Host: example.com\r\n\r\nhello
                 ↑
            headers_end

因此请求体的起始位置是 headers_end + 4.

解析请求行

HTTP/1.x 请求行的最小格式是:

METHOD SP TARGET SP HTTP-VERSION CRLF

先找到两个空格, 再把空格和行尾的 \r 替换为 \0, 三个字段就变成了可以直接使用的 C 字符串:

/** 解析请求行, 直接修改原始请求行缓冲区, 将字段间空格以及请求行结尾替换为 '\0', 分割为独立的 C 字符串, 保存到 request
 * - `line`: 请求行的起始地址
 * - `line_end`: 请求行的结束位置
 * - `request`: 保存解析结果
 */
static int parse_request_line(char *line, char *line_end, struct http_request *request) {
    char *method_end = memchr(      // 查找第 1 个空格, 用于定位 HTTP 方法(method)的结束位置
        line,
        ' ',
        (size_t)(line_end - line)
    );

    if (method_end == NULL || method_end == line) {
        return -1;
    }

    char *target = method_end + 1;  // 第 1 个空格之后是请求目标(target)的起始位置
    char *target_end = memchr(      // 查找下一个空格, 用于定位请求目标的结束位置
        target,
        ' ',
        (size_t)(line_end - target)
    );

    if (target_end == NULL || target_end == target) {
        return -1;
    }

    char *version = target_end + 1; // 第 2 个空格之后是 HTTP 版本号的起始位置
    if (version == line_end) {
        return -1;
    }

    /** 将字段之间的分隔符替换为字符串结束符 */
    *method_end = '\0';
    *target_end = '\0';
    *line_end = '\0';

    request->method = line;
    request->target = target;
    request->version = version;

    /** 仅处理 HTTP/1.x */
    if (strcmp(request->version, "HTTP/1.0") != 0 && strcmp(request->version, "HTTP/1.1") != 0) {
        return -1;
    }

    return 0;
}

例如:

GET /index.html HTTP/1.1\r\n
│   │           │
│   │           └── version
│   └── target
└── method

这里没有解析 URL 的 query string, 也没有判断 method 是否具有某种业务含义. parser 只负责把 HTTP/1.x 请求行切分成三个字段, 并确认 version 是本例支持的 HTTP/1.0 或 HTTP/1.1.

解析请求头

请求头的每一行都可以用第一个 : 分成 name 和 value. 最后一个 header 的 \r\n 正好也是 \r\n\r\n 的前两个字节, 所以循环遍历到 headers_end 即可:

/** 解析请求头
 * - `headers_start`: 请求头区域的起始地址
 * - `headers_end`: 请求头区域的结束位置
 * - `request`: 保存解析结果
 */
static int parse_request_headers(char *headers_start, char *headers_end, struct http_request *request) {
    char *line = headers_start;

    while (line < headers_end) {
        char *line_end = find_crlf(line, headers_end + 2);  // 查找当前请求头行的 CRLF 结束位置, headers_end + 2 用于确保查找范围包含最后一行的 CRLF
        if (line_end == NULL || line_end > headers_end) {
            return -1;
        }

        char *colon = memchr(   // 在当前请求头行中查找 ':'，用于分隔请求头名称和值
            line,
            ':',
            (size_t)(line_end - line)
        );

        if (colon == NULL || colon == line) {
            return -1;
        }

        if (request->header_count >= HTTP_MAX_HEADERS) {    // 检查请求头数量是否超过预定义的最大限制
            return -1;
        }

        *colon = '\0';
        *line_end = '\0';

        char *value = colon + 1;                    // ':' 后面的内容是请求头的值
        while (*value == ' ' || *value == '\t') {   // 跳过请求头值前的可选空格或制表符
            value++;
        }

        /** 以索引保存当前请求头的名称和值 */
        request->headers[request->header_count].name = line;
        request->headers[request->header_count].value = value;
        request->header_count++;

        if (line_end == headers_end) {
            break;
        }

        line = line_end + 2;
    }

    return 0;
}

例如 Content-Length: 5\r\n 在 buffer 中会变成:

Content-Length\0 5\0
│              │
│              └── value
└── name

这里沿用前文的简化假设: header 使用常见写法, 不处理大小写转换、header folding 和非法 header. 解析器只保存 header, 不在这里解释 Host、Content-Type 等业务含义.

记录请求体

HTTP/1.x parser 只需要确定 body 的起始位置和长度, 不应该继续猜测 body 是 JSON、表单还是其他格式. recv_http1_request() 已经根据 Content-Length 收到了完整请求, 因此这里把头部结束后的剩余 bytes 作为 body:

这里的"剩余 bytes"成立, 是因为本例已经限定为"没有 body"或"使用 Content-Length 的 body". 如果请求使用 Transfer-Encoding: chunked, 头部后面的 bytes 还需要经过 chunk decoder, 不能直接当作最终 body.

/** 记录请求体的起始位置和长度, 不把 body 当作字符串处理 */
static int parse_request_body(char *buf, size_t length, char *headers_end, struct http_request *request) {
    size_t body_start = (size_t)(headers_end - buf) + 4;

    if (body_start > length) {
        return -1;
    }

    request->body = buf + body_start;
    request->body_length = length - body_start;
    return 0;
}

body_length 可能为 0. 如果 body 中包含 \0, 仍然可以通过 body_length 正确处理:

fwrite(request.body, 1, request.body_length, stdout);

组合解析函数

现在由外层函数负责按顺序调用三个 parser:

/** 将一条完整的 HTTP/1.x request bytes 解析为 struct http_request */
int parse_http1_request(
    char *buf,
    size_t length,
    struct http_request *request
) {
    char *end;
    char *line_end;
    char *headers_end;

    if (buf == NULL || request == NULL || length == 0) {
        return -1;
    }

    end = buf + length;
    headers_end = find_headers_end(buf, end);
    if (headers_end == NULL) {
        return -1;
    }

    /* 请求行是整段 bytes 中的第一行 */
    line_end = find_crlf(buf, end);
    if (line_end == NULL || line_end > headers_end) {
        return -1;
    }

    memset(request, 0, sizeof *request);

    if (parse_request_line(buf, line_end, request) == -1) {
        return -1;
    }

    if (parse_request_headers(line_end + 2, headers_end, request) == -1) {
        return -1;
    }

    if (parse_request_body(buf, length, headers_end, request) == -1) {
        return -1;
    }

    return 0;
}

调用方式:

struct http_request parsed_request;

if (request_length > 0 && parse_http1_request(
    request,
    (size_t)request_length,
    &parsed_request
) == -1) {
    fprintf(stderr, "invalid HTTP request\n");
} else if (request_length > 0) {
    printf("%s %s %s\n",
        parsed_request.method,
        parsed_request.target,
        parsed_request.version
    );

    for (size_t i = 0; i < parsed_request.header_count; i++) {
        printf("%s: %s\n",
            parsed_request.headers[i].name,
            parsed_request.headers[i].value
        );
    }

    printf("body bytes: %zu\n", parsed_request.body_length);
}

这里传入的是 recv_http1_request() 返回的 request_length, 不是 sizeof request, 也不是 strlen(request). 前者是 buffer 容量, 后者无法正确表示可能包含 \0 的原始 body bytes.

整个处理流程可以概括为:

recv_http1_request()
    -> 收到完整 request bytes
    -> 找到第一行和 \r\n\r\n
    -> parse_request_line()
    -> parse_request_headers()
    -> parse_request_body()
    -> 得到 struct http_request

HTTP/1.x 请求响应函数

前面已经获取并解析了完整的 HTTP/1.x request. 接下来服务器需要按照 HTTP/1.x 的格式组织 response, 再通过 connected socket 发送给客户端.

响应格式

HTTP/1.x response 通常由以下部分组成:

状态行 + 响应头 + 空行 + 响应体

一个返回 Hello World! 的最小 response 可以写成:

HTTP/1.1 200 OK\r\n
Content-Type: text/plain\r\n
Content-Length: 12\r\n
Connection: close\r\n
\r\n
Hello World!
  • 状态行 HTTP/1.1 200 OK 表示使用 HTTP/1.1, 状态码为 200, reason phrase 为 OK.
  • Content-Type 表示响应体是普通文本.
  • Content-Length 表示响应体的 byte 数量. Hello World! 是 12 个 ASCII bytes, 不包括 C 字符串末尾的 \0.
  • Connection: close 表示响应发送完成后关闭连接, 本例不处理 persistent connection.
  • 状态行、响应头和空行都必须使用 \r\n, 响应头结束后再额外发送一个 \r\n.

请求和响应使用相同的 HTTP/1.x 分隔规则, 但两者第一行的含义不同:

request:  METHOD TARGET HTTP-VERSION\r\n
response: HTTP-VERSION STATUS-CODE REASON-PHRASE\r\n

函数设计

本例的 response 内容固定为 Hello World!, 不根据 target 或请求头执行不同业务, 所以函数只需要接收已连接的 socket:

sockfd
   ↓
send_http1_response()
   ↓
HTTP/1.x response bytes

函数接口可以设计为:

#include <sys/types.h>

/** 向已连接的 socket 发送一个固定的 HTTP/1.x response
 * - `sockfd`: 已连接的套接字文件描述符
 * 成功返回 0, 发送失败返回 -1
 */
int send_http1_response(int sockfd);

这里没有把 struct http_request 作为参数传入, 是因为当前 response 与 request 内容无关. 实际服务器通常会根据 request->method、request->target 等字段生成不同 response, 但那属于后续的业务逻辑.

具体实现

#include <errno.h>
#include <sys/socket.h>

/** 向已连接的 socket 发送一个固定的 HTTP/1.x response */
int send_http1_response(int sockfd) {
    static const char response[] =
        "HTTP/1.1 200 OK\r\n"
        "Content-Type: text/plain\r\n"
        "Content-Length: 12\r\n"
        "Connection: close\r\n"
        "\r\n"
        "Hello World!";

    size_t sent = 0;
    size_t response_length = sizeof response - 1;    // 不计算 C 字符串末尾的 '\0'

    while (sent < response_length) {
        ssize_t bytes_sent = send(
            sockfd,
            response + sent,          // 从尚未发送的 byte 继续发送
            response_length - sent,
            0
        );

        if (bytes_sent > 0) {
            sent += (size_t)bytes_sent;
            continue;
        }

        if (bytes_sent == -1 && errno == EINTR) {
            continue;
        }

        return -1;
    }

    return 0;
}

相邻的 C 字符串字面量会在编译时自动连接, 因此上面的 response 是一段连续的 bytes. 每个 \r\n 都是 HTTP/1.x 协议中的换行符, 不能写成只包含 \n 的普通文本换行.

send() 返回的 byte 数量可能小于请求发送的数量, 所以使用 sent 记录进度, 直到整个 response 都被发送. 如果系统调用因为信号中断并设置 errno 为 EINTR, 重新调用即可. 这里不使用 strlen(response), 而是使用 sizeof response - 1 得到这段固定数组的实际 byte 数量.

组合请求与响应

在一个连接上, 可以把前面得到的 request parser 和当前的 response function 串起来:

struct http_request parsed_request;

if (request_length > 0) {
    if (parse_http1_request(
        request,
        (size_t)request_length,
        &parsed_request
    ) == -1) {
        fprintf(stderr, "invalid HTTP request\n");
    } else if (send_http1_response(connfd) == -1) {
        perror("send_http1_response");
    }
}

close(connfd);

当前教学示例的完整处理流程是:

recv_http1_request()
    -> 收到完整 request bytes
    -> parse_http1_request()
    -> 得到 struct http_request
    -> send_http1_response()
    -> close(connfd)

客户端收到的不是单独的 Hello World!, 而是完整的 HTTP response. HTTP client 先读取状态行和响应头, 根据 Content-Length: 12 读取后面的 12 个 body bytes, 然后因为 Connection: close 知道服务器会关闭这个连接.

本例只返回固定的成功 response, 没有处理非法请求、不同的 status code、不同的资源和 persistent connection. 这些功能可以在理解 request/response 的基本格式后再逐步加入.
