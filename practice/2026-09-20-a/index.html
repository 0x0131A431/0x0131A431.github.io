问题

现有的 极简服务器 启动后, 只要客户端连接到服务器并发起一次 HTTP GET 请求，无论请求路径是什么，都会返回同一个固定响应:

HTTP/1.1 200 OK
Content-Type: text/plain
Content-Length: 12
Connection: close

Hello World!

本次需改进

GET /                   -> 200 OK
GET /hello              -> 200 OK
GET /<anything-else>    -> 404 Not Found

思路

源码已实现获取完整的 HTTP 报文, 只需要让 response 层使用已有的 request->target 并判断

  1. http/http_response.c
#include "http_response.h"

#include <string.h>
#include <sys/socket.h>

void send_http1_response(int sockfd, struct http_request *request) {
    const char *response;

    static const char ok_response[] =
        "HTTP/1.1 200 OK\r\n"
        "Content-Type: text/plain\r\n"
        "Content-Length: 12\r\n"
        "Connection: close\r\n"
        "\r\n"
        "Hello World!";

    /** 404 Response */
    static const char not_found_response[] =
        "HTTP/1.1 404 Not Found\r\n"
        "Content-Type: text/plain\r\n"
        "Content-Length: 9\r\n"
        "Connection: close\r\n"
        "\r\n"
        "Not Found";

    /** 判断请求路径 */
    if (strcmp(request->target, "/") == 0 || strcmp(request->target, "/hello") == 0) {
        response = ok_response;
    }
    else {
        response = not_found_response;
    }

    size_t sent = 0;
    size_t response_length = strlen(response);  // 使用 strlen 计算长度

    while (sent < response_length) {
        sent += (size_t)send(
            sockfd,
            response + sent,
            response_length - sent,
            0
        );
    }
}
  1. main.c
#include <netdb.h>
#include <sys/socket.h>
#include <unistd.h>

#include "http_parse.h"
#include "http_recv.h"
#include "http_response.h"

int main(void) {
    struct addrinfo hints = {
        .ai_family = AF_UNSPEC,
        .ai_socktype = SOCK_STREAM,
        .ai_flags = AI_PASSIVE
    };
    struct addrinfo *serverinfo;
    getaddrinfo(NULL, "3490", &hints, &serverinfo);
    int listenfd = socket(
        serverinfo->ai_family,
        serverinfo->ai_socktype,
        serverinfo->ai_protocol
    );
    bind(listenfd, serverinfo->ai_addr, serverinfo->ai_addrlen);
    freeaddrinfo(serverinfo);
    listen(listenfd, 5);

    while (1) {
        int connectfd = accept(listenfd, NULL, NULL);

        /**************** HTTP ****************/
        char buf[8192];
        struct http_request request;
        size_t request_length = recv_http1_request(connectfd, buf, sizeof buf);
        parse_http1_request(buf, request_length, &request);

        /** 传入 request */
        send_http1_response(connectfd, &request);

        /**************************************/

        close(connectfd);
    }
}
